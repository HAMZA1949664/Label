   115→/** Carte animée à l'apparition (entrées douces en cascade). */
   116→function CarteAnimee({
   117→  children,
   118→  delai = 0,
   119→}: {
   120→  children: React.ReactNode;
   121→  delai?: number;
   122→}) {
   123→  return (
   124→    <motion.div
   125→      initial={{ opacity: 0, y: 14 }}
   126→      animate={{ opacity: 1, y: 0 }}
   127→      transition={{ duration: 0.4, delay: delai, ease: "easeOut" }}
   128→    >
   129→      {children}
   130→    </motion.div>
   131→  );
   132→}

   134→/** Télécharge un blob côté navigateur. */
   135→function telecharger(blob: Blob, filename: string) {
   136→  const url = URL.createObjectURL(blob);
   137→  const a = document.createElement("a");
   138→  a.href = url;
   139→  a.download = filename;
   140→  document.body.appendChild(a);
   141→  a.click();
   142→  document.body.removeChild(a);
   143→  URL.revokeObjectURL(url);
   144→}

   527→                Clément Design

   570→            <CarteAnimee delai={0}>

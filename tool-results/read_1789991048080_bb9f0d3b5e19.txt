   133→/** Carte animée à l'apparition (entrées douces en cascade). */
   134→function CarteAnimee({
   135→  children,
   136→  delai = 0,
   137→}: {
   138→  children: React.ReactNode;
   139→  delai?: number;
   140→}) {
   141→  return (
   142→    <motion.div
   143→      initial={{ opacity: 0, y: 14 }}
   144→      animate={{ opacity: 1, y: 0 }}
   145→      transition={{ duration: 0.4, delay: delai, ease: "easeOut" }}
   146→    >
   147→      {children}
   148→    </motion.div>
   149→  );
   150→}

   152→/** Télécharge un blob côté navigateur. */
   153→function telecharger(blob: Blob, filename: string) {
   154→  const url = URL.createObjectURL(blob);
   155→  const a = document.createElement("a");
   156→  a.href = url;
   157→  a.download = filename;
   158→  document.body.appendChild(a);
   159→  a.click();
   160→  document.body.removeChild(a);
   161→  URL.revokeObjectURL(url);
   162→}

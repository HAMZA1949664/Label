   121→/** « il y a 3 min » … */
   122→function relatif(iso: string | null): string {
   123→  if (!iso) return "—";
   124→  const t = new Date(iso).getTime();
   125→  if (Number.isNaN(t)) return "—";
   126→  const s = Math.max(0, Math.round((Date.now() - t) / 1000));
   127→  if (s < 45) return "à l'instant";
   128→  const m = Math.round(s / 60);
   129→  if (m < 60) return `il y a ${m} min`;
   130→  const h = Math.round(m / 60);
   131→  if (h < 24) return `il y a ${h} h`;
   132→  const j = Math.round(h / 24);
   133→  return `il y a ${j} j`;
   134→}
   135→
   136→/** Date + heure locale (fr-FR). */
   137→function horodatage(iso: string | null): string {
   138→  if (!iso) return "—";
   139→  const d = new Date(iso);
   140→  if (Number.isNaN(d.getTime())) return "—";
   141→  return d.toLocaleString("fr-FR", { dateStyle: "short", timeStyle: "short" });
   142→}

   147→  if (statut.dernierTest.ok === false) return "bg-red-400";
   148→  return "bg-emerald-400";
   149→}

   230→        variant: "destructive",
   231→        title: "Lecture du journal cloud impossible (réseau).",

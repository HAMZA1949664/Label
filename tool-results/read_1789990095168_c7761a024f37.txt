   116→   116→function CarteAnimee({
   117→   117→  children,
   118→   118→  delai = 0,
   119→   119→}: {
   120→   120→  children: React.ReactNode;
   121→   121→  delai?: number;
   122→   122→}) {
   123→   123→  return (
   124→   124→    <motion.div
   125→   125→      initial={{ opacity: 0, y: 14 }}
   126→   126→      animate={{ opacity: 1, y: 0 }}
   127→   127→      transition={{ duration: 0.4, delay: delai, ease: "easeOut" }}
   128→   128→    >
   129→   129→      {children}
   130→   130→    </motion.div>
   131→   131→  );
   132→   132→}

   134→   134→/** Télécharge un blob côté navigateur. */
   135→   135→function telecharger(blob: Blob, filename: string) {
   136→   136→  const url = URL.createObjectURL(blob);
   137→   137→  const a = document.createElement("a");
   138→   138→  a.href = url;
   139→   139→  a.download = filename;
   140→   140→  document.body.appendChild(a);
   141→   141→  a.click();
   142→   142→  document.body.removeChild(a);
   143→   143→  URL.revokeObjectURL(url);
   144→   144→}
